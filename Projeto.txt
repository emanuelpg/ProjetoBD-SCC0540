-- ------------------------------------------------------------
-- 1. Dominios (tipos com regra de validacao reaproveitavel)
-- ------------------------------------------------------------

-- CPF: 11 digitos, somente numeros (sem pontos e traco)
CREATE DOMAIN dom_cpf AS CHAR(11)
    CHECK (VALUE ~ '^[0-9]{11}$');

-- CNPJ: 14 digitos, somente numeros
CREATE DOMAIN dom_cnpj AS CHAR(14)
    CHECK (VALUE ~ '^[0-9]{14}$');

-- CEP: 8 digitos, somente numeros
CREATE DOMAIN dom_cep AS CHAR(8)
    CHECK (VALUE ~ '^[0-9]{8}$');

-- E-mail: formato basico texto@texto.texto
CREATE DOMAIN dom_email AS VARCHAR(120)
    CHECK (VALUE ~* '^[^@[:space:]]+@[^@[:space:]]+\.[^@[:space:]]+$');

-- Telefone: DDD + numero, 10 ou 11 digitos
CREATE DOMAIN dom_telefone AS VARCHAR(11)
    CHECK (VALUE ~ '^[0-9]{10,11}$');

-- Unidade federativa
CREATE DOMAIN dom_uf AS CHAR(2)
    CHECK (VALUE IN ('AC','AL','AP','AM','BA','CE','DF','ES','GO','MA',
                     'MT','MS','MG','PA','PB','PR','PE','PI','RJ','RN',
                     'RS','RO','RR','SC','SP','SE','TO'));

-- Valores monetarios nao negativos
CREATE DOMAIN dom_dinheiro AS NUMERIC(12,2)
    CHECK (VALUE >= 0);

-- Papel exercido por um participante credenciado em uma sessao
CREATE TYPE papel_inscricao AS ENUM ('ouvinte', 'instrutor', 'palestrante');

-- EVENTO -----------------------------------------------------
CREATE TABLE evento (
    EventoId   INTEGER GENERATED BY DEFAULT AS IDENTITY,
    Nome       VARCHAR(150) NOT NULL,
    Inicio     TIMESTAMP    NOT NULL,
    Fim        TIMESTAMP    NOT NULL,
    Descricao  TEXT,

    CONSTRAINT pk_evento PRIMARY KEY (EventoId),
    CONSTRAINT ck_evento_periodo CHECK (Fim > Inicio)
);

-- PROMOTOR (entidade generica da especializacao disjunta) -----
CREATE TABLE promotor (
    PromotorId INTEGER GENERATED BY DEFAULT AS IDENTITY,
    Email      dom_email,
    Telefone   dom_telefone,
    Nome       VARCHAR(150) NOT NULL,
    CEP        dom_cep,
    Endereco   VARCHAR(200),

    CONSTRAINT pk_promotor          PRIMARY KEY (PromotorId),
    CONSTRAINT uq_promotor_email    UNIQUE (Email),
    CONSTRAINT uq_promotor_telefone UNIQUE (Telefone)
);

-- PESSOA_PROMOTOR (subtipo: pessoa fisica) --------------------
-- A PK e tambem FK para promotor (heranca da chave).
CREATE TABLE pessoa_promotor (
    PessoaPId  INTEGER  NOT NULL,
    CPF        dom_cpf  NOT NULL,

    CONSTRAINT pk_pessoa_promotor     PRIMARY KEY (PessoaPId),
    CONSTRAINT uq_pessoa_promotor_cpf UNIQUE (CPF),
    CONSTRAINT fk_pessoa_promotor_promotor
        FOREIGN KEY (PessoaPId) REFERENCES promotor (PromotorId)
        ON DELETE CASCADE ON UPDATE CASCADE
);

-- ORGANIZACAO (subtipo: pessoa juridica) ----------------------
CREATE TABLE organizacao (
    OrganizacaoId INTEGER  NOT NULL,
    CNPJ          dom_cnpj NOT NULL,

    CONSTRAINT pk_organizacao      PRIMARY KEY (OrganizacaoId),
    CONSTRAINT uq_organizacao_cnpj UNIQUE (CNPJ),
    CONSTRAINT fk_organizacao_promotor
        FOREIGN KEY (OrganizacaoId) REFERENCES promotor (PromotorId)
        ON DELETE CASCADE ON UPDATE CASCADE
);

-- PATROCINIO (Evento M:N Promotor) ----------------------------
CREATE TABLE patrocinio (
    EventoId        INTEGER      NOT NULL,
    PromotorId      INTEGER      NOT NULL,
    Valor           dom_dinheiro NOT NULL,
    Data_patrocinio DATE         NOT NULL DEFAULT CURRENT_DATE,

    CONSTRAINT pk_patrocinio PRIMARY KEY (EventoId, PromotorId),
    CONSTRAINT fk_patrocinio_evento
        FOREIGN KEY (EventoId) REFERENCES evento (EventoId)
        ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT fk_patrocinio_promotor
        FOREIGN KEY (PromotorId) REFERENCES promotor (PromotorId)
        ON DELETE RESTRICT ON UPDATE CASCADE
);

-- ATIVIDADE (Evento 1:N Atividade, participacao total) --------
CREATE TABLE atividade (
    AtividadeId INTEGER GENERATED BY DEFAULT AS IDENTITY,
    EventoId    INTEGER      NOT NULL,
    Nome        VARCHAR(150) NOT NULL,
    Descricao   TEXT,

    CONSTRAINT pk_atividade PRIMARY KEY (AtividadeId),
    CONSTRAINT fk_atividade_evento
        FOREIGN KEY (EventoId) REFERENCES evento (EventoId)
        ON DELETE CASCADE ON UPDATE CASCADE
);

-- ESPACO -----------------------------------------------------
CREATE TABLE espaco (
    EspacoId      INTEGER GENERATED BY DEFAULT AS IDENTITY,
    Nome          VARCHAR(100) NOT NULL,
    LimitePessoas INTEGER      NOT NULL,
    CEP           dom_cep      NOT NULL,
    Numero        VARCHAR(10)  NOT NULL,
    Rua           VARCHAR(150),
    Bairro        VARCHAR(100),
    Cidade        VARCHAR(100),
    Estado        dom_uf,

    CONSTRAINT pk_espaco PRIMARY KEY (EspacoId),
    CONSTRAINT ck_espaco_limite CHECK (LimitePessoas > 0)
);

-- SESSAO (Atividade 1:N Sessao, participacao total;
--         Sessao N:1 Espaco, participacao parcial) -----------
CREATE TABLE sessao (
    SessaoId       INTEGER GENERATED BY DEFAULT AS IDENTITY,
    AtividadeId    INTEGER      NOT NULL,
    EspacoId       INTEGER,                    -- pode ficar sem espaco definido
    Nome           VARCHAR(150) NOT NULL,
    DataHoraInicio TIMESTAMP    NOT NULL,
    Duracao        INTEGER      NOT NULL,      -- em minutos
    Vagas          INTEGER      NOT NULL,
    Valor          dom_dinheiro NOT NULL DEFAULT 0,

    -- Atributo gerado: termino da sessao, calculado a partir do inicio
    -- e da duracao. Facilita a verificacao de conflito de horario.
    DataHoraFim    TIMESTAMP GENERATED ALWAYS AS
                   (DataHoraInicio + make_interval(mins => Duracao)) STORED,

    CONSTRAINT pk_sessao PRIMARY KEY (SessaoId),
    CONSTRAINT ck_sessao_duracao CHECK (Duracao > 0),
    CONSTRAINT ck_sessao_vagas   CHECK (Vagas > 0),
    CONSTRAINT fk_sessao_atividade
        FOREIGN KEY (AtividadeId) REFERENCES atividade (AtividadeId)
        ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT fk_sessao_espaco
        FOREIGN KEY (EspacoId) REFERENCES espaco (EspacoId)
        ON DELETE SET NULL ON UPDATE CASCADE
);

-- RECURSOS ---------------------------------------------------
CREATE TABLE recursos (
    RecursoId INTEGER GENERATED BY DEFAULT AS IDENTITY,
    Nome      VARCHAR(100) NOT NULL,
    Descricao TEXT         NOT NULL,

    CONSTRAINT pk_recursos      PRIMARY KEY (RecursoId),
    CONSTRAINT uq_recursos_nome UNIQUE (Nome)
);

-- ESTOQUE (Espaco M:N Recursos) ------------------------------
CREATE TABLE estoque (
    EstoqueId  INTEGER GENERATED BY DEFAULT AS IDENTITY,
    EspacoId   INTEGER NOT NULL,
    RecursoId  INTEGER NOT NULL,
    Quantidade INTEGER NOT NULL DEFAULT 1,

    CONSTRAINT pk_estoque PRIMARY KEY (EstoqueId),
    CONSTRAINT uq_estoque_espaco_recurso UNIQUE (EspacoId, RecursoId),
    CONSTRAINT ck_estoque_quantidade CHECK (Quantidade > 0),
    CONSTRAINT fk_estoque_espaco
        FOREIGN KEY (EspacoId) REFERENCES espaco (EspacoId)
        ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT fk_estoque_recurso
        FOREIGN KEY (RecursoId) REFERENCES recursos (RecursoId)
        ON DELETE RESTRICT ON UPDATE CASCADE
);

-- RECURSOS_SESSAO (Sessao M:N Recursos) ----------------------
CREATE TABLE recursos_sessao (
    RecursoSessaoId INTEGER GENERATED BY DEFAULT AS IDENTITY,
    SessaoId        INTEGER NOT NULL,
    RecursoId       INTEGER NOT NULL,
    QtdNecessaria   INTEGER NOT NULL DEFAULT 1,

    CONSTRAINT pk_recursos_sessao PRIMARY KEY (RecursoSessaoId),
    CONSTRAINT uq_recursos_sessao_sessao_recurso UNIQUE (SessaoId, RecursoId),
    CONSTRAINT ck_recursos_sessao_qtd CHECK (QtdNecessaria > 0),
    CONSTRAINT fk_recursos_sessao_sessao
        FOREIGN KEY (SessaoId) REFERENCES sessao (SessaoId)
        ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT fk_recursos_sessao_recurso
        FOREIGN KEY (RecursoId) REFERENCES recursos (RecursoId)
        ON DELETE RESTRICT ON UPDATE CASCADE
);

-- PARTICIPANTE -----------------------------------------------
CREATE TABLE participante (
    ParticipanteId INTEGER GENERATED BY DEFAULT AS IDENTITY,
    CPF            dom_cpf      NOT NULL,
    Nome           VARCHAR(150) NOT NULL,
    DataNascimento DATE         NOT NULL,

    CONSTRAINT pk_participante     PRIMARY KEY (ParticipanteId),
    CONSTRAINT uq_participante_cpf UNIQUE (CPF),
    CONSTRAINT ck_participante_nascimento CHECK (DataNascimento > DATE '1900-01-01')
);

-- CREDENCIAMENTO (Participante M:N Evento - agregacao) --------
-- Organizacao = TRUE indica que o participante e organizador do evento.
CREATE TABLE credenciamento (
    CredenciamentoId INTEGER GENERATED BY DEFAULT AS IDENTITY,
    ParticipanteId   INTEGER NOT NULL,
    EventoId         INTEGER NOT NULL,
    Organizacao      BOOLEAN NOT NULL DEFAULT FALSE,

    CONSTRAINT pk_credenciamento PRIMARY KEY (CredenciamentoId),
    CONSTRAINT uq_credenciamento_participante_evento UNIQUE (ParticipanteId, EventoId),
    CONSTRAINT fk_credenciamento_participante
        FOREIGN KEY (ParticipanteId) REFERENCES participante (ParticipanteId)
        ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT fk_credenciamento_evento
        FOREIGN KEY (EventoId) REFERENCES evento (EventoId)
        ON DELETE CASCADE ON UPDATE CASCADE
);

-- INSCRICAO (Credenciamento M:N Sessao) ----------------------
CREATE TABLE inscricao (
    InscricaoId      INTEGER GENERATED BY DEFAULT AS IDENTITY,
    CredenciamentoId INTEGER         NOT NULL,
    SessaoId         INTEGER         NOT NULL,
    DataHora         TIMESTAMP       NOT NULL DEFAULT CURRENT_TIMESTAMP,
    Papel            papel_inscricao NOT NULL DEFAULT 'ouvinte',

    CONSTRAINT pk_inscricao PRIMARY KEY (InscricaoId),
    CONSTRAINT uq_inscricao_credenciamento_sessao UNIQUE (CredenciamentoId, SessaoId),
    CONSTRAINT fk_inscricao_credenciamento
        FOREIGN KEY (CredenciamentoId) REFERENCES credenciamento (CredenciamentoId)

    CONSTRAINT fk_inscricao_sessao
        FOREIGN KEY (SessaoId) REFERENCES sessao (SessaoId)
        ON DELETE CASCADE ON UPDATE CASCADE

-- ------------------------------------------------------------
-- Trigger 0: Conflito de espaco
-- Duas sessoes nao podem ocupar o mesmo espaco no mesmo intervalo.
-- ------------------------------------------------------------
CREATE OR REPLACE FUNCTION fn_valida_conflito_espaco()
RETURNS TRIGGER AS $$
DECLARE
    v_conflito INTEGER;
BEGIN
    -- Sessao sem espaco definido nao conflita com nada
    IF NEW.EspacoId IS NULL THEN
        RETURN NEW;
    END IF;

    -- Serializa agendamentos concorrentes no mesmo espaco.
    -- Advisory lock (e nao FOR UPDATE em espaco) para nao entrar em conflito
    -- com o FOR SHARE de trg_sessao_requisitos, que roda antes.
    PERFORM pg_advisory_xact_lock(NEW.EspacoId);

    SELECT COUNT(*)
      INTO v_conflito
      FROM sessao
     WHERE EspacoId = NEW.EspacoId
       AND SessaoId <> COALESCE(NEW.SessaoId, -1)
       AND NEW.DataHoraInicio < DataHoraFim
       AND (NEW.DataHoraInicio + make_interval(mins => NEW.Duracao)) > DataHoraInicio;

    IF v_conflito > 0 THEN
        RAISE EXCEPTION 'Erro de agendamento: o espaco % ja possui outra sessao alocada neste horario.',
            NEW.EspacoId
            USING ERRCODE = 'check_violation';
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_valida_conflito_espaco ON sessao;

CREATE TRIGGER trg_valida_conflito_espaco
BEFORE INSERT OR UPDATE OF EspacoId, DataHoraInicio, Duracao ON sessao
FOR EACH ROW
EXECUTE FUNCTION fn_valida_conflito_espaco();


-- ------------------------------------------------------------
-- Trigger 1: Conflito de horario na inscricao
-- A mesma pessoa nao pode ter inscricoes em sessoes sobrepostas
-- ------------------------------------------------------------
CREATE OR REPLACE FUNCTION fn_verifica_conflito_inscricao()
RETURNS TRIGGER AS $$
DECLARE
    v_participante INTEGER;
    v_inicio       TIMESTAMP;
    v_fim          TIMESTAMP;
    v_conf_nome    VARCHAR(150);
    v_conf_inicio  TIMESTAMP;
    v_conf_fim     TIMESTAMP;
BEGIN
    -- Descobre quem e a pessoa por tras do credenciamento
    SELECT ParticipanteId
      INTO v_participante
      FROM credenciamento
     WHERE CredenciamentoId = NEW.CredenciamentoId;

    -- Trava o participante para serializar inscricoes concorrentes da mesma pessoa
    PERFORM 1
       FROM participante
      WHERE ParticipanteId = v_participante
        FOR UPDATE;

    -- Intervalo da sessao que esta sendo inscrita
    SELECT DataHoraInicio, DataHoraFim
      INTO v_inicio, v_fim
      FROM sessao
     WHERE SessaoId = NEW.SessaoId;

    -- Procura qualquer outra inscricao da mesma pessoa cujo intervalo se sobreponha
    SELECT s.Nome, s.DataHoraInicio, s.DataHoraFim
      INTO v_conf_nome, v_conf_inicio, v_conf_fim
      FROM inscricao i
      JOIN credenciamento c ON c.CredenciamentoId = i.CredenciamentoId
      JOIN sessao s         ON s.SessaoId         = i.SessaoId
     WHERE c.ParticipanteId = v_participante
       AND (TG_OP = 'INSERT' OR i.InscricaoId <> NEW.InscricaoId)
       AND s.DataHoraInicio < v_fim
       AND v_inicio         < s.DataHoraFim
     LIMIT 1;

    IF FOUND THEN
        RAISE EXCEPTION
            'Conflito de horario: o participante % ja esta inscrito na sessao "%" (% a %).',
            v_participante, v_conf_nome,
            to_char(v_conf_inicio, 'DD/MM/YYYY HH24:MI'),
            to_char(v_conf_fim,    'DD/MM/YYYY HH24:MI')
            USING ERRCODE = 'check_violation';
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_inscricao_conflito_horario ON inscricao;

CREATE TRIGGER trg_inscricao_conflito_horario
BEFORE INSERT OR UPDATE OF CredenciamentoId, SessaoId ON inscricao
FOR EACH ROW
EXECUTE FUNCTION fn_verifica_conflito_inscricao();


-- ------------------------------------------------------------
-- Trigger 2: Limite de vagas (inscricao com sessao)
-- Somente ouvintes ocupam vaga.
-- ------------------------------------------------------------
CREATE OR REPLACE FUNCTION fn_verifica_limite_vagas()
RETURNS TRIGGER AS $$
DECLARE
    v_vagas    INTEGER;
    v_ocupadas INTEGER;
BEGIN
    -- Somente ouvintes ocupam vaga: outros papeis passam direto
    IF NEW.Papel <> 'ouvinte' THEN
        RETURN NEW;
    END IF;

    -- Trava a linha da sessao para serializar inscricoes concorrentes
    -- (sem isso, duas transacoes simultaneas podem estourar o limite).
    SELECT Vagas
      INTO v_vagas
      FROM sessao
     WHERE SessaoId = NEW.SessaoId
       FOR UPDATE;

    -- Conta os ouvintes ja inscritos
    -- (no UPDATE, ignora a propria linha que esta sendo alterada).
    SELECT COUNT(*)
      INTO v_ocupadas
      FROM inscricao
     WHERE SessaoId = NEW.SessaoId
       AND Papel = 'ouvinte'
       AND (TG_OP = 'INSERT' OR InscricaoId <> NEW.InscricaoId);

    IF v_ocupadas >= v_vagas THEN
        RAISE EXCEPTION 'Sessao % esgotada: % de % vagas preenchidas.',
            NEW.SessaoId, v_ocupadas, v_vagas
            USING ERRCODE = 'check_violation';
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_inscricao_limite_vagas ON inscricao;

-- Papel incluido: instrutor/palestrante que vira ouvinte passa a ocupar vaga
CREATE TRIGGER trg_inscricao_limite_vagas
BEFORE INSERT OR UPDATE OF SessaoId, Papel ON inscricao
FOR EACH ROW
EXECUTE FUNCTION fn_verifica_limite_vagas();


-- ------------------------------------------------------------
-- Trigger 3 (em SESSAO): capacidade do espaco + recursos ja exigidos
-- ------------------------------------------------------------
CREATE OR REPLACE FUNCTION fn_verifica_requisitos_sessao()
RETURNS TRIGGER AS $$
DECLARE
    v_limite   INTEGER;
    v_recurso  VARCHAR(100);
    v_exigido  INTEGER;
    v_estoque  INTEGER;
BEGIN
    -- Sessao sem espaco definido nao tem requisitos a validar
    IF NEW.EspacoId IS NULL THEN
        RETURN NEW;
    END IF;

    -- FOR SHARE evita que o espaco seja alterado/removido durante a validacao
    SELECT LimitePessoas
      INTO v_limite
      FROM espaco
     WHERE EspacoId = NEW.EspacoId
       FOR SHARE;

    -- (a) Vagas da sessao nao podem exceder a capacidade do espaco
    IF NEW.Vagas > v_limite THEN
        RAISE EXCEPTION
            'Sessao "%": % vagas excedem o limite de % pessoas do espaco %.',
            NEW.Nome, NEW.Vagas, v_limite, NEW.EspacoId
            USING ERRCODE = 'check_violation';
    END IF;

    -- (b) Recursos que a sessao ja exige devem existir no estoque do espaco
    --     (no INSERT ainda nao ha recursos_sessao; vale para troca de espaco)
    SELECT r.Nome, rs.QtdNecessaria, COALESCE(e.Quantidade, 0)
      INTO v_recurso, v_exigido, v_estoque
      FROM recursos_sessao rs
      JOIN recursos r        ON r.RecursoId = rs.RecursoId
      LEFT JOIN estoque e    ON e.EspacoId  = NEW.EspacoId
                            AND e.RecursoId = rs.RecursoId
     WHERE rs.SessaoId = NEW.SessaoId
       AND COALESCE(e.Quantidade, 0) < rs.QtdNecessaria
     LIMIT 1;

    IF FOUND THEN
        RAISE EXCEPTION
            'Espaco % nao atende a sessao "%": recurso "%" exige % unidade(s), mas ha %.',
            NEW.EspacoId, NEW.Nome, v_recurso, v_exigido, v_estoque
            USING ERRCODE = 'check_violation';
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_sessao_requisitos ON sessao;

CREATE TRIGGER trg_sessao_requisitos
BEFORE INSERT OR UPDATE OF EspacoId, Vagas ON sessao
FOR EACH ROW
EXECUTE FUNCTION fn_verifica_requisitos_sessao();


-- ------------------------------------------------------------
-- Trigger 4 (em RECURSOS_SESSAO): o espaco da sessao tem o recurso?
-- ------------------------------------------------------------
CREATE OR REPLACE FUNCTION fn_verifica_recurso_sessao()
RETURNS TRIGGER AS $$
DECLARE
    v_espaco     INTEGER;
    v_disponivel INTEGER;
BEGIN
    -- FOR SHARE impede que o espaco da sessao mude durante a validacao
    SELECT EspacoId
      INTO v_espaco
      FROM sessao
     WHERE SessaoId = NEW.SessaoId
       FOR SHARE;

    -- Sessao sem espaco: nada a validar (sera validado quando o espaco for definido)
    IF v_espaco IS NULL THEN
        RETURN NEW;
    END IF;

    SELECT Quantidade
      INTO v_disponivel
      FROM estoque
     WHERE EspacoId  = v_espaco
       AND RecursoId = NEW.RecursoId;

    IF v_disponivel IS NULL THEN
        RAISE EXCEPTION
            'O espaco % nao possui o recurso % no estoque.',
            v_espaco, NEW.RecursoId
            USING ERRCODE = 'check_violation';
    ELSIF v_disponivel < NEW.QtdNecessaria THEN
        RAISE EXCEPTION
            'Recurso % insuficiente no espaco %: necessario %, disponivel %.',
            NEW.RecursoId, v_espaco, NEW.QtdNecessaria, v_disponivel
            USING ERRCODE = 'check_violation';
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_recursos_sessao_requisitos ON recursos_sessao;

CREATE TRIGGER trg_recursos_sessao_requisitos
BEFORE INSERT OR UPDATE OF SessaoId, RecursoId, QtdNecessaria ON recursos_sessao
FOR EACH ROW
EXECUTE FUNCTION fn_verifica_recurso_sessao();
